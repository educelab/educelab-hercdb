# Create node(:Educelab ID) and write its UUIP as a property
LOAD CSV WITH HEADERS FROM "file:///MellonUUID_mod.csv" as line
WITH line
MERGE (e:EduceLabID {id:line.EduceLab_ID, uuid:line.UUID})


# Create PHerc node 
LOAD CSV WITH HEADERS FROM "file:///MellonUUID_mod.csv" as line
WITH line
WHERE line.PHerc IS NOT NULL
MERGE (p:PHerc {name: line.PHerc})

# Create Pezzo_Cr node and attach it to PHerc node
# Without the PHerc number as an attribute, the every cornice #1 becomes the same 
# Think about this to see if there is a better way
LOAD CSV WITH HEADERS FROM "file:///MellonUUID_mod.csv" as line
WITH line
WHERE line.Pezzo_Cr IS NOT NULL
MATCH (p:PHerc {name: line.PHerc})
MERGE (c:CornicePezzo {pherc_name: line.PHerc, cor_name: line.Pezzo_Cr})
MERGE (p)-[:HAS]->(c)


# Create PHerc, and Pezzo/Cr nodes and connect Cornice to EduceLab ID
LOAD CSV WITH HEADERS FROM "file:///MellonUUID_mod.csv" as line
WITH line
WHERE line.Pezzo_Cr IS NOT NULL
MATCH (e:EduceLabID{id: line.EduceLab_ID})
MATCH (c:CornicePezzo {pherc_name: line.PHerc, cor_name: line.Pezzo_Cr})
MERGE (c)-[:ASSIGNED_TO]->(e)

# Deal with a missing Pezzo_Cr case
LOAD CSV WITH HEADERS FROM "file:///MellonUUID_mod.csv" as line
WITH line
WHERE (line.Pezzo_Cr IS NULL) AND (line.PHerc IS NOT NULL)
MATCH (e:EduceLabID{id: line.EduceLab_ID})
MATCH (p:PHerc {name: line.PHerc})
MERGE (p)-[:ASSIGNED_TO]->(e)


# Create Replacement UUID
LOAD CSV WITH HEADERS FROM "file:///MellonUUID_mod.csv" as line
WITH line
WHERE line.Replacement_UUID IS NOT NULL
MATCH (e:EduceLabID{id: line.EduceLab_ID})
MERGE (r:ReplacementUUID {id: line.Replacement_UUID})
MERGE (e)-[:REPLACED_BY]->(r)
