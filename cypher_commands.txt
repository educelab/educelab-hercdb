######### For Mellon UUID sheet ####################
## Change the headers to 
## "EduceLab_ID,UUID,PHerc,Pezzo_Cr,Replacement_UUID"
## (Replace blank spaces and slash with underscores) 


# Create node(:Educelab ID) and write its UUIP as a property
LOAD CSV WITH HEADERS FROM "file:///MellonUUID_mod.csv" as line
WITH line
MERGE (e:EduceLabID {id:line.EduceLab_ID, uuid:line.UUID})


# Record which UUID was replaced by which
LOAD CSV WITH HEADERS FROM "file:///MellonUUID_mod.csv" as line
WITH line
WHERE line.Replacement_UUID IS NOT NULL
MATCH (orig:EduceLabID{id: line.EduceLab_ID})
MATCH (new:EduceLabID{uuid: line.Replacement_UUID})
MERGE (orig)<-[:REPLACES]-(new)


# Create PHerc node 
LOAD CSV WITH HEADERS FROM "file:///MellonUUID_mod.csv" as line
WITH line
WHERE line.PHerc IS NOT NULL
MERGE (p:PHerc {name: line.PHerc})


# Create Cornice node and attach it to PHerc node and EduceLabID node
# For now, we will call them all "Cornice" even if they are pezzi
LOAD CSV WITH HEADERS FROM "file:///MellonUUID_mod.csv" as line
WITH line
WHERE line.Pezzo_Cr IS NOT NULL
MATCH (p:PHerc {name: line.PHerc})
MATCH (e:EduceLabID {id: line.EduceLab_ID})
MERGE (p)-[:HAS]->(:Cornice {name: line.Pezzo_Cr})<-[:ASSIGNED_TO]-(e)

######## negatives_mod.csv #############
## Changed headers to
## Image_num,Negatives_series,PHerc,cornice,UUID,storage_loc


# Create a FlatbedScanDataset node
LOAD CSV WITH HEADERS FROM "file:///negatives_mod.csv" as line
WITH line
WHERE line.Image_num IS NOT NULL
MATCH (e:EduceLabID {uuid: line.UUID})
MERGE (e)<-[:BELONGS_TO]-(:FlatbedScanDataset {img_num: line.Image_num, neg_series: coalesce(line.Negatives_series, "unknown"), neg_storage: coalesce(line.storage_loc, "unknown")})

# by creating a dummy EduceLabID Node
LOAD CSV WITH HEADERS FROM "file:///metadata.csv" as line
WITH line
WHERE (line.UUID IS NOT NULL) AND (line.CorniceNum IS NOT NULL)
MATCH (:PHerc {name: line.PapyrusNum})-[:HAS]->(c:CornicePezzo {name: line.CorniceNum})
